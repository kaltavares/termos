O termo foi gerado.
' + 'Clique aqui para abrir o PDF' + ''; return HtmlService.createHtmlOutput(htmlContent) .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL); } catch (erro) { var htmlErro = '' + '' + '' + 'Erro ao processar: ' + erro.message + ''; return HtmlService.createHtmlOutput(htmlErro) .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL); } }
